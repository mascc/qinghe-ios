  function evpBytesToKey(pass, salt, len) {
    const p = new TextEncoder().encode(pass);
    let d = new Uint8Array(0), prev = new Uint8Array(0);
    while (d.length < len) {
      const inp = new Uint8Array(prev.length + p.length + salt.length);
      inp.set(prev, 0); inp.set(p, prev.length); inp.set(salt, prev.length + p.length);
      prev = md5(inp);
      const nd = new Uint8Array(d.length + prev.length);
      nd.set(d, 0); nd.set(prev, d.length); d = nd;
    }
    return d.slice(0, len);
  }

  function b64ToBytes(b64) {
    const bin = atob(b64);
    const u = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) u[i] = bin.charCodeAt(i);
    return u;
  }

  async function decryptOpenSSL(b64, pass) {
    if (!(window.crypto && crypto.subtle)) throw new Error('no SubtleCrypto');
    const buf = b64ToBytes(b64);
    if (String.fromCharCode.apply(null, buf.slice(0, 8)) !== 'Salted__') throw new Error('not salted');
    const salt = buf.slice(8, 16), data = buf.slice(16);
    const ki = evpBytesToKey(pass, salt, 48);
    const key = ki.slice(0, 32), iv = ki.slice(32, 48);
    const ck = await crypto.subtle.importKey('raw', key, { name: 'AES-CBC' }, false, ['decrypt']);
    const pt = await crypto.subtle.decrypt({ name: 'AES-CBC', iv }, ck, data);
    return new TextDecoder().decode(pt);
  }

  function injectCss() {
    if (document.getElementById('hx-style')) return;
    const s = document.createElement('style');
    s.id = 'hx-style';
    s.textContent = CSS;
    (document.head || document.documentElement).appendChild(s);
  }

  /* ---------------- 播放器构建 ---------------- */

  function buildPlayer(wrapper, id, line) {
    injectCss();
    const root = document.createElement('div');
    root.className = 'hx-root';
    root.innerHTML = `
      <video class="hx-video" playsinline webkit-playsinline preload="metadata"></video>
      <div class="hx-center">${ICON.play}</div>
      <div class="hx-spin"></div>
      <div class="hx-top">
        <div class="hx-title"></div>
        <div class="hx-tools">
          <button class="hx-btn" data-act="line">${ICON.line}<span>线路</span></button>
          <button class="hx-btn icon" data-act="refresh" title="刷新">${ICON.refresh}</button>
          <button class="hx-btn icon" data-act="close" title="关闭">${ICON.close}</button>
        </div>
      </div>
      <div class="hx-bar">
        <input class="hx-seek" type="range" min="0" max="1000" value="0" step="1">
        <div class="hx-row">
          <button class="hx-btn icon" data-act="play">${ICON.play}</button>
          <button class="hx-btn icon" data-act="back" title="-10s">${ICON.back}</button>
          <button class="hx-btn icon" data-act="fwd" title="+10s">${ICON.fwd}</button>
          <span class="hx-time">0:00 / 0:00</span>
          <span class="hx-spacer"></span>
          <button class="hx-btn icon" data-act="mute">${ICON.vol}</button>
          <input class="hx-vol" type="range" min="0" max="100" value="100">
          <button class="hx-btn" data-act="rate">${ICON.rate}<span>1.0x</span></button>
          <button class="hx-btn icon" data-act="fs" title="全屏/全面">${ICON.fs}</button>
          <button class="hx-btn icon" data-act="mini" title="缩小悬浮">${ICON.mini}</button>
        </div>
      </div>
      <div class="hx-pop hx-ratelist"></div>
      <div class="hx-pop hx-linelist"></div>
      <div class="hx-toast"></div>`;
    wrapper.appendChild(root);

    const video = root.querySelector('.hx-video');
    const seek = root.querySelector('.hx-seek');
    const vol = root.querySelector('.hx-vol');
    const timeEl = root.querySelector('.hx-time');
    const titleEl = root.querySelector('.hx-title');
    const playBtn = root.querySelector('[data-act="play"]');
    const muteBtn = root.querySelector('[data-act="mute"]');
    const rateLabel = root.querySelector('[data-act="rate"] span');

    const st = {
      id, line, root, video, wrapper,
      dragging: false, hideTimer: null, recovering: false,
      mode: 'overlay', lastTap: 0
    };

    const wrap = wrapper.querySelector('.premium-overlay');
    if (wrap) wrap.style.display = 'none';
    wrapper.querySelectorAll('video').forEach(v => { try { v.pause(); } catch (e) {} });

    root.addEventListener('click', e => {
      const btn = e.target.closest('[data-act]');
      if (!btn) return;
      const act = btn.dataset.act;
      if (act === 'play') togglePlay();
      else if (act === 'back') jump(-10);
      else if (act === 'fwd') jump(10);
      else if (act === 'mute') toggleMute();
      else if (act === 'fs') toggleFs();
      else if (act === 'mini') toggleMini();
      else if (act === 'close') closeByUser();
      else if (act === 'refresh') reload(true);
      else if (act === 'rate') togglePop('.hx-ratelist');
      else if (act === 'line') togglePop('.hx-linelist');
    });

    root.querySelector('.hx-center').addEventListener('click', togglePlay);

    function togglePop(sel) {
      const pop = root.querySelector(sel);
      const show = !pop.classList.contains('show');
      root.querySelectorAll('.hx-pop').forEach(p => p.classList.remove('show'));
      if (show) pop.classList.add('show');
    }

    const rateList = root.querySelector('.hx-ratelist');
    CFG.rates.forEach(r => {
      const b = document.createElement('button');
      b.textContent = r + 'x';
      b.dataset.rate = r;
      if (r === 1) b.classList.add('on');
      b.addEventListener('click', () => {
        video.playbackRate = r;
        rateLabel.textContent = r + 'x';
        rateList.querySelectorAll('button').forEach(x => x.classList.toggle('on', x.dataset.rate == r));
        rateList.classList.remove('show');
        try { localStorage.setItem('hx_rate', r); } catch (e) {}
      });
      rateList.appendChild(b);
    });

    const lineList = root.querySelector('.hx-linelist');
    CFG.lines.forEach(l => {
      const b = document.createElement('button');
      b.textContent = l.n;
      b.dataset.line = l.k;
      if (l.k === line) b.classList.add('on');
      b.addEventListener('click', () => {
        lineList.classList.remove('show');
        if (l.k === st.line) return;
        st.line = l.k;
        lineList.querySelectorAll('button').forEach(x => x.classList.toggle('on', x.dataset.line === l.k));
        reload(true);
      });
      lineList.appendChild(b);
    });

    video.addEventListener('play', () => { root.classList.add('hx-playing'); playBtn.innerHTML = ICON.pause; scheduleHide(); });
    video.addEventListener('pause', () => { root.classList.remove('hx-playing'); playBtn.innerHTML = ICON.play; showUI(); });
    video.addEventListener('waiting', () => root.classList.add('hx-loading'));
    video.addEventListener('playing', () => root.classList.remove('hx-loading'));
    video.addEventListener('canplay', () => root.classList.remove('hx-loading'));
    video.addEventListener('timeupdate', () => {
      if (!st.dragging) seek.value = video.duration ? (video.currentTime / video.duration) * 1000 : 0;
      timeEl.textContent = fmt(video.currentTime) + ' / ' + fmt(video.duration);
    });
    video.addEventListener('volumechange', () => {
      vol.value = video.muted ? 0 : video.volume * 100;
      muteBtn.innerHTML = (video.muted || video.volume === 0) ? ICON.mute : ICON.vol;
    });
    let tapTimer = null;
    video.addEventListener('click', () => {
      if (tapTimer) { clearTimeout(tapTimer); tapTimer = null; toggleFs(); }
      else { tapTimer = setTimeout(() => { tapTimer = null; togglePlay(); }, 280); }
    });
    video.addEventListener('error', () => { if (!video._hls) onPlaybackError({}); });

    const startDrag = () => { st.dragging = true; };
    const endDrag = () => {
      if (!st.dragging) return;
      st.dragging = false;
      if (video.duration) video.currentTime = (seek.value / 1000) * video.duration;
    };
    seek.addEventListener('pointerdown', startDrag);
    seek.addEventListener('touchstart', startDrag, { passive: true });
    seek.addEventListener('input', () => {
      if (video.duration) timeEl.textContent = fmt((seek.value / 1000) * video.duration) + ' / ' + fmt(video.duration);
    });
    seek.addEventListener('change', endDrag);
    seek.addEventListener('pointerup', endDrag);
    seek.addEventListener('touchend', endDrag);
    vol.addEventListener('input', () => { video.muted = false; video.volume = vol.value / 100; });

    function togglePlay() { if (video.paused) video.play().catch(() => {}); else video.pause(); }
    function jump(d) { video.currentTime = Math.min(Math.max((video.currentTime || 0) + d, 0), video.duration || 1e9); }
    function toggleMute() { video.muted = !video.muted; }

    function showUI() { root.classList.remove('hx-hideui'); clearTimeout(st.hideTimer); }
    function scheduleHide() {
      clearTimeout(st.hideTimer);
      if (st.mode === 'mini') return;
      st.hideTimer = setTimeout(() => { if (!video.paused) root.classList.add('hx-hideui'); }, 2800);
    }
    root.addEventListener('pointermove', showUI);
    root.addEventListener('touchstart', showUI, { passive: true });

    function toggleFs() {
      const isFs = root.classList.contains('hx-fs');
      if (!isFs) {
        st.mode = 'fs';
        root.classList.remove('hx-mini');
        root.classList.add('hx-fs');
        document.body.style.overflow = 'hidden';
        const fn = root.requestFullscreen || root.webkitRequestFullscreen;
        if (fn) { try { fn.call(root); } catch (e) {} }
      } else {
        st.mode = 'overlay';
        root.classList.remove('hx-fs');
        document.body.style.overflow = '';
        if (document.fullscreenElement) { try { document.exitFullscreen(); } catch (e) {} }
      }
      setTimeout(fit, 120);
    }

    function toggleMini() {
      if (root.classList.contains('hx-mini')) {
        st.mode = 'overlay';
        root.classList.remove('hx-mini');
        root.style.left = ''; root.style.top = ''; root.style.right = ''; root.style.bottom = '';
      } else {
        st.mode = 'mini';
        root.classList.remove('hx-fs', 'hx-hideui');
        if (document.fullscreenElement) { try { document.exitFullscreen(); } catch (e) {} }
        root.classList.add('hx-mini');
      }
      setTimeout(fit, 120);
    }

    function onFsChange() {
      if (!document.fullscreenElement && root.classList.contains('hx-fs') && st.mode === 'fs') {
        root.classList.remove('hx-fs');
        st.mode = 'overlay';
        document.body.style.overflow = '';
        setTimeout(fit, 100);
      }
    }
    document.addEventListener('fullscreenchange', onFsChange);

    function fit() {
      const v = video;
      v.style.width = '100%';
      v.style.height = '100%';
    }

    st.toggleFs = toggleFs;
    st.reload = reload;
    st.destroy = destroy;

    async function reload(showSpin) {
      if (showSpin) root.classList.add('hx-loading');
      try {
        const r = await resolveM3u8(st.id, st.line);
        st.title = r.title || '';
        titleEl.textContent = r.title || '';
        await attach(video, r.url);
        applyRate();
        video.play().catch(() => {});
      } catch (e) {
        toast('获取视频失败：' + (e && e.message ? e.message : e));
      } finally {
        root.classList.remove('hx-loading');
      }
    }

    function applyRate() {
      let r = parseFloat(localStorage.getItem('hx_rate') || '1');
      if (CFG.rates.indexOf(r) < 0) r = 1;
      video.playbackRate = r;
      rateLabel.textContent = r + 'x';
      rateList.querySelectorAll('button').forEach(x => x.classList.toggle('on', parseFloat(x.dataset.rate) === r));
    }

    function destroy() {
      if (video._hls) { try { video._hls.destroy(); } catch (e) {} }
      try { video.pause(); } catch (e) {}
      if (document.fullscreenElement) { try { document.exitFullscreen(); } catch (e) {} }
      document.body.style.overflow = '';
      root.remove();
      if (wrapper.querySelector('.premium-overlay')) wrapper.querySelector('.premium-overlay').style.display = '';
      state = null;
    }

    function closeByUser() {
      dismissed.add(st.id);
      destroy();
      const w = document.querySelector('.player-wrapper');
      if (w && videoIdFromUrl() === st.id) addTrigger(w, st.id);
    }

    state = st;
    reload(false);
    return st;
  }

  /* ---------------- 触发/接管 ---------------- */

  function addTrigger(wrapper, id) {
    if (wrapper.querySelector('.hx-trigger')) return;
    const b = document.createElement('button');
    b.className = 'hx-trigger';
    b.textContent = '解锁播放';
    b.addEventListener('click', () => {
      b.style.display = 'none';
      dismissed.delete(id);
      buildPlayer(wrapper, id, 'default1');
    });
    wrapper.appendChild(b);
  }

  function maybeInit() {
    const id = videoIdFromUrl();
    if (!id) return;
    if (state && state.id === id && document.body.contains(state.root)) return;

    const wrapper = document.querySelector('.player-wrapper');
    if (!wrapper) return;

    if (state && state.id !== id) { try { state.destroy(); } catch (e) {} }

    const masked = !!wrapper.querySelector('.premium-overlay');
    if (masked) {
      if (!wrapper.querySelector('.hx-root')) {
        if (!dismissed.has(id)) buildPlayer(wrapper, id, 'default1');
        else addTrigger(wrapper, id);
      }
    } else {
      addTrigger(wrapper, id);
    }
  }

  function boot() {
    if (booted) return;
    booted = true;
    const _push = history.pushState, _replace = history.replaceState;
    history.pushState = function () { _push.apply(this, arguments); setTimeout(maybeInit, 60); };
    history.replaceState = function () { _replace.apply(this, arguments); setTimeout(maybeInit, 60); };
    window.addEventListener('popstate', () => setTimeout(maybeInit, 60));

    const obs = new MutationObserver(() => {
      if (maybeInit._t) return;
      maybeInit._t = setTimeout(() => { maybeInit._t = null; maybeInit(); }, 250);
    });
    obs.observe(document.documentElement, { childList: true, subtree: true });
    maybeInit();
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', boot);
  else boot();
})();
